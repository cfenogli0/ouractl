# octl mcp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `octl mcp` serves MCP over stdio with three tools (`query`, `status`, `sync`) as thin adapters over the existing `export`, `store`, and `sync` packages.

**Architecture:** New `internal/mcp` package builds an `*mcp.Server` from a `Deps` struct (store, fetcher-factory closure, enabled datasets, clock). The CLI command owns config loading and client construction; `internal/mcp` never imports `oura` or `auth`. Domain failures are returned as `IsError` tool results via a `toolError` helper, never as handler errors, so the model always sees them.

**Tech Stack:** Go 1.26, `github.com/modelcontextprotocol/go-sdk/mcp` (official SDK, v1.x), existing `cobra` CLI, `modernc.org/sqlite` store.

**Spec:** `docs/superpowers/specs/2026-07-14-octl-mcp-design.md`

## Global Constraints

- Module: `github.com/cfenogli0/ouractl`; work on branch `feat/octl-mcp`.
- `internal/mcp` may import `store`, `export`, `sync`, `datasets` only. NOT `oura`, NOT `auth`, NOT `config`.
- All tool failures return `IsError: true` results with a human-readable message. Handlers never return a non-nil Go error for domain failures.
- `query` output must be produced by `export.Export` with `Format: "json"` (byte-identical to `octl export`).
- Row cap default 200; exceeding it is an explicit error naming the row count. Never truncate silently.
- Auth (the fetcher factory) is called only inside the `sync` handler. `query`/`status` must work when the factory errors.
- Dates are `YYYY-MM-DD` strings everywhere, validated with `time.Parse("2006-01-02", ...)`.
- Tests use `mcp.NewInMemoryTransports()`; no live Oura API, no real auth.
- Commit after every task with the exact message given. Every commit ends with `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.

```go
res, _ := cs.CallTool(ctx, &mcp.CallToolParams{Name: "query", Arguments: map[string]any{...}})
res.IsError                                        // tool-level failure
res.Content[0].(*mcp.TextContent).Text             // text payload
```

```go
// Package mcp serves MCP tools (query, status, sync) over the octl store.
// It is a thin adapter: query formatting is export.Export, freshness is
// store.Stats, sync is sync.Run. Client/auth construction stays in the CLI
// and arrives here only as an opaque fetcher factory, so read-only sessions
// never touch auth.
package mcp

import (
	"fmt"

	"github.com/cfenogli0/ouractl/internal/datasets"
	"github.com/cfenogli0/ouractl/internal/store"
	syncpkg "github.com/cfenogli0/ouractl/internal/sync"
	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
)

const defaultRowLimit = 200

// Deps is everything the server needs. FetcherFactory is called lazily, only
// by the sync tool, so a broken token store cannot take down read-only use.
type Deps struct {
	Store          *store.Store
	FetcherFactory func() (syncpkg.Fetcher, error)
	Enabled        []datasets.Dataset // sync default set (config collection.enabled)
	OverlapDays    int
	Today          func() string // YYYY-MM-DD; injectable for tests
	Version        string
}

// New builds the MCP server with the query, status, and sync tools registered.
func New(d Deps) *sdk.Server {
	server := sdk.NewServer(&sdk.Implementation{Name: "octl", Version: d.Version}, nil)
	registerTools(server, d)
	return server
}

// namesToDatasets resolves dataset names against the registry, erroring on
// unknown names with the valid list so a typo fails loud.
func namesToDatasets(names []string) ([]datasets.Dataset, error) {
	out := make([]datasets.Dataset, 0, len(names))
	for _, n := range names {
		ds, ok := datasets.Get(n)
		if !ok {
			return nil, fmt.Errorf("unknown dataset %q (valid: %v)", n, datasets.All())
		}
		out = append(out, ds)
	}
	return out, nil
}
```

```go
package mcp

import (
	"context"
	"encoding/json"
	"fmt"
	"strings"

	sdk "github.com/modelcontextprotocol/go-sdk/mcp"
)

// toolError returns a model-visible tool failure. Domain failures always take
// this path; a non-nil Go error from a handler is reserved for SDK/protocol
// plumbing.
func toolError(format string, a ...any) *sdk.CallToolResult {
	return &sdk.CallToolResult{
		IsError: true,
		Content: []sdk.Content{&sdk.TextContent{Text: fmt.Sprintf(format, a...)}},
	}
}

func textResult(s string) *sdk.CallToolResult {
	return &sdk.CallToolResult{Content: []sdk.Content{&sdk.TextContent{Text: s}}}
}

func registerTools(server *sdk.Server, d Deps) {
	registerStatus(server, d)
}

type statusIn struct{}

type statusRow struct {
	Dataset       string `json:"dataset"`
	Rows          int    `json:"rows"`
	LastSyncedDay string `json:"last_synced_day"`
	LastRunAt     string `json:"last_run_at"`
}

func registerStatus(server *sdk.Server, d Deps) {
	desc := "Per-dataset freshness of the local Oura store: row count, last synced day, " +
		"last sync run time. Call this first when data recency matters; if the store is " +
		"stale, say so or run the sync tool. Datasets: " + strings.Join(allDatasetNames(), ", ") + "."
	sdk.AddTool(server, &sdk.Tool{Name: "status", Description: desc},
		func(ctx context.Context, req *sdk.CallToolRequest, in statusIn) (*sdk.CallToolResult, any, error) {
			stats, err := d.Store.Stats()
			if err != nil {
				return toolError("read store stats: %v", err), nil, nil
			}
			rows := make([]statusRow, 0, len(stats))
			for _, st := range stats {
				rows = append(rows, statusRow{
					Dataset: st.Dataset, Rows: st.Rows,
					LastSyncedDay: st.LastSyncedDay, LastRunAt: st.LastRunAt,
				})
			}
			out, err := json.Marshal(rows)
			if err != nil {
				return toolError("encode status: %v", err), nil, nil
			}
			return textResult(string(out)), nil, nil
		})
}

func allDatasetNames() []string {
	return dsAll()
}
```

```go
type queryIn struct {
	Dataset string   `json:"dataset" jsonschema:"dataset name, one of the octl datasets"`
	From    string   `json:"from" jsonschema:"start day inclusive, YYYY-MM-DD"`
	To      string   `json:"to" jsonschema:"end day inclusive, YYYY-MM-DD"`
	Fields  []string `json:"fields,omitempty" jsonschema:"optional top-level fields to keep; omit for full records"`
	Limit   int      `json:"limit,omitempty" jsonschema:"max rows returned, default 200; raise deliberately for big ranges"`
}

func registerQuery(server *sdk.Server, d Deps) {
	desc := "Query the local Oura store (SQLite, synced longitudinal record; not the live API). " +
		"Returns a JSON array, one object per record, same shape as `octl export`. " +
		"Use fields to keep results small. Datasets: " + strings.Join(allDatasetNames(), ", ") + "."
	sdk.AddTool(server, &sdk.Tool{Name: "query", Description: desc},
		func(ctx context.Context, req *sdk.CallToolRequest, in queryIn) (*sdk.CallToolResult, any, error) {
			if _, ok := dsGet(in.Dataset); !ok {
				return toolError("unknown dataset %q; valid: %s", in.Dataset, strings.Join(allDatasetNames(), ", ")), nil, nil
			}
			for _, day := range []string{in.From, in.To} {
				if !validDay(day) {
					return toolError("bad date %q: want YYYY-MM-DD", day), nil, nil
				}
			}
			limit := in.Limit
			if limit <= 0 {
				limit = defaultRowLimit
			}
			recs, err := d.Store.QueryRange(in.Dataset, in.From, in.To)
			if err != nil {
				return toolError("query store: %v", err), nil, nil
			}
			if len(recs) > limit {
				return toolError("%d rows match %s [%s, %s], over the limit of %d; narrow the range, project fewer days, or pass a higher limit",
					len(recs), in.Dataset, in.From, in.To, limit), nil, nil
			}
			if len(recs) == 0 {
				wm, err := d.Store.Watermark(in.Dataset)
				if err != nil {
					return toolError("read watermark: %v", err), nil, nil
				}
				if wm == "" {
					return toolError("dataset %s has never been synced; run the sync tool or `octl sync`", in.Dataset), nil, nil
				}
				return textResult("[]"), nil, nil
			}
			var buf strings.Builder
			opts := exportOptions(in)
			if err := doExport(d.Store, opts, &buf); err != nil {
				return toolError("export: %v", err), nil, nil
			}
			return textResult(strings.TrimSpace(buf.String())), nil, nil
		})
}
```

```go
type syncIn struct {
	Datasets []string `json:"datasets,omitempty" jsonschema:"datasets to sync; omit for the configured enabled set"`
	Days     int      `json:"days,omitempty" jsonschema:"backfill the last N days instead of incremental-from-watermark"`
}

type syncRow struct {
	Dataset       string `json:"dataset"`
	RowsAdded     int    `json:"rows_added"`
	TotalRows     int    `json:"total_rows"`
	LastSyncedDay string `json:"last_synced_day"`
}

func registerSync(server *sdk.Server, d Deps) {
	desc := "Fetch fresh data from the Oura API into the local store. Spends real API " +
		"rate-limit budget; use when status shows stale data, not before every query. " +
		"Returns per-dataset rows_added (new rows this run), total_rows, last_synced_day."
	sdk.AddTool(server, &sdk.Tool{Name: "sync", Description: desc},
		func(ctx context.Context, req *sdk.CallToolRequest, in syncIn) (*sdk.CallToolResult, any, error) {
			sets := d.Enabled
			if len(in.Datasets) > 0 {
				var err error
				sets, err = namesToDatasets(in.Datasets)
				if err != nil {
					return toolError("%v", err), nil, nil
				}
			}
			before, err := statsByDataset(d.Store)
			if err != nil {
				return toolError("read store stats: %v", err), nil, nil
			}
			fetcher, err := d.FetcherFactory()
			if err != nil {
				return toolError("auth: %v", err), nil, nil
			}
			now := d.Today()
			opts := syncOptions(in, now, d.OverlapDays)
			if in.Days > 0 {
				start, err := minusDays(now, in.Days)
				if err != nil {
					return toolError("resolve days: %v", err), nil, nil
				}
				opts.Start = start
			}
			if err := runSync(ctx, d.Store, fetcher, sets, opts); err != nil {
				return toolError("sync: %v", err), nil, nil
			}
			after, err := statsByDataset(d.Store)
			if err != nil {
				return toolError("read store stats after sync: %v", err), nil, nil
			}
			rows := make([]syncRow, 0, len(sets))
			for _, ds := range sets {
				a := after[ds.Name]
				rows = append(rows, syncRow{
					Dataset:       ds.Name,
					RowsAdded:     a.Rows - before[ds.Name].Rows,
					TotalRows:     a.Rows,
					LastSyncedDay: a.LastSyncedDay,
				})
			}
			out, err := json.Marshal(rows)
			if err != nil {
				return toolError("encode sync result: %v", err), nil, nil
			}
			return textResult(string(out)), nil, nil
		})
}
```

```go
import ( /* add */ "context" )

func statsByDataset(s *store.Store) (map[string]store.DatasetStat, error) {
	stats, err := s.Stats()
	if err != nil {
		return nil, err
	}
	m := make(map[string]store.DatasetStat, len(stats))
	for _, st := range stats {
		m[st.Dataset] = st
	}
	return m, nil
}

func minusDays(day string, n int) (string, error) { return syncpkg.MinusDays(day, n) }

func syncOptions(in syncIn, now string, overlap int) syncpkg.Options {
	source := "sync"
	if in.Days > 0 {
		source = "backfill"
	}
	return syncpkg.Options{End: now, Now: now, OverlapDays: overlap, Source: source}
}

func runSync(ctx context.Context, s *store.Store, f syncpkg.Fetcher, sets []datasets.Dataset, opts syncpkg.Options) error {
	return syncpkg.Run(ctx, s, f, sets, opts)
}
```

`rows_added` is a row-count diff, not an upsert count: overlap-window re-upserts of existing rows do not change the count, and new rows do. That is the number the model actually wants ("what did this sync bring in").

```go
package cli

import (
	"github.com/cfenogli0/ouractl/internal/mcp"
	"github.com/cfenogli0/ouractl/internal/store"
	syncpkg "github.com/cfenogli0/ouractl/internal/sync"

	"github.com/spf13/cobra"
)

func newMcpCmd(g *globalFlags, version string) *cobra.Command {
	return &cobra.Command{
		Use:   "mcp",
		Short: "Serve MCP over stdio (query, status, sync tools over the local store)",
		RunE: func(cmd *cobra.Command, _ []string) error {
			c, err := g.load()
			if err != nil {
				return err
			}
			if err := mustValidate(c); err != nil {
				return err
			}
			sets, err := enabledDatasets(c.Collection.Enabled)
			if err != nil {
				return err
			}
			s, err := store.Open(c.Storage.Path)
			if err != nil {
				return err
			}
			defer s.Close()
			server := mcp.New(mcp.Deps{
				Store: s,
				// Lazy: only the sync tool pays for auth; a broken token
				// store must not take down read-only sessions.
				FetcherFactory: func() (syncpkg.Fetcher, error) { return buildClient(c) },
				Enabled:        sets,
				OverlapDays:    c.Collection.OverlapDays,
				Today:          today,
				Version:        version,
			})
			return server.Run(cmd.Context(), &sdk.StdioTransport{})
		},
	}
}
```
